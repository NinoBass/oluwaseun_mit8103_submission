FROM customer_order o
JOIN customer c ON c.customer_id = o.customer_id
WHERE o.order_status = 'PAID'
  AND o.order_date < TIMESTAMPTZ '2026-10-31 00:00+01' - INTERVAL '24 hours'
ORDER BY o.order_date
LIMIT 50;
